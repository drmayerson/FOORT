28public:

31
32 Metric(bool rlogscale = false);
33
34 // Basic functions that return the metric with indices down or up:
35 // pure virtual as they must be defined in the descendant classes.
36 //
37 // Get the metric at Point p, indices down
38 virtual TwoIndex getMetric_dd(const Point &p) const = 0;
39 // Get the metric at Point p, indices up
40 virtual TwoIndex getMetric_uu(const Point &p) const = 0;
41
42 // The following functions return the Christoffel and other derivative quantities of the metric.
43 // They are implemented for this base class, BUT are left as virtual functions to allow for

46 //

51 // Get the Kretschmann scalar
52 virtual real getKretschmann(const Point &p) const;
53
54 // Function to get the description of the metric
55 // (used for outputting to the screen while running and possibly to the output files)

57 virtual std::string getFullDescriptionStr() const;

59 bool getrLogScale() const;

61protected:
