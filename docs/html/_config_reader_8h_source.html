173template <class OutputType>
174bool ConfigReader::ConfigCollection::LookupValue(int SettingIndex, OutputType &theOutput) const
175{
176 // Get the setting value, if it is indeed of this type
177 const OutputType *OutputPointer{std::get_if<OutputType>(&m_Settings[SettingIndex].SettingValue)};
178 if (OutputPointer)
179 {
180 // Success, the setting is indeed of this type; set the output accordingly
181 theOutput = *OutputPointer;
182 return true;
183 }
184 return false;
185}
