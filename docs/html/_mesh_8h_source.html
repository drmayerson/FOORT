163 // - maxPixels: max. nr of pixels that can be integrated in TOTAL, over all iterations (if 0, then this is infinite,
164 // i.e. we keep integrating until all squares are maximally subdivided or have weight 0)
165 // - initialPixels: initial number of pixels to integrate (spaced equally over the screen)
166 // - maxSubdivide: maximum number of times that we can subdivide squares (note: 1 denotes the
167 // initial grid, so 2 would be subdividing the squares once)
168 // - iterationPixels: maximum number of pixels to subdivide in each integration iteration
169 // (max number of pixels that will be integrates is then 5*iterationPixels)
170 // - initialSubToFinal: once we decide to subdivide a square, do we automatically keep subdividing it
171 // until we reach maxSubdivision?
172 // - valdiag: the "value" and "distance" Diagnostic to use
173 SquareSubdivisionMesh(largecounter maxPixels, largecounter initialPixels, int maxSubdivide, largecounter iterationPixels, bool initialSubToFinal,
174 DiagBitflag valdiag)
175 : m_InitialPixels{static_cast<pixelcoord>(sqrt(initialPixels)) * static_cast<pixelcoord>(sqrt(initialPixels))},
176 m_MaxSubdivide{maxSubdivide},
177 m_RowColumnSize{static_cast<pixelcoord>((sqrt(initialPixels) - 1) * ExpInt(2, maxSubdivide - 1) + 1)},
178 m_PixelsLeft{maxPixels}, m_MaxPixels{maxPixels}, m_InfinitePixels{maxPixels == 0}, m_IterationPixels{iterationPixels},
179 m_InitialSubDividideToFinal{initialSubToFinal}, Mesh(valdiag)
180 {
181 if constexpr (dimension != 4)
182 ScreenOutput("SquareSubdivisionMesh only defined in 4D!", OutputLevel::Level_0_WARNING);

186
187 // Initialize the initial square, equally spaced grid
