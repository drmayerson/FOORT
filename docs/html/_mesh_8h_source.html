165 // Constructor must be called with arguments:
166 // - maxPixels: max. nr of pixels that can be integrated in TOTAL, over all iterations (if 0, then this is infinite,
167 // i.e. we keep integrating until all squares are maximally subdivided or have weight 0)
168 // - initialPixels: initial number of pixels to integrate (spaced equally over the screen)
169 // - maxSubdivide: maximum number of times that we can subdivide squares (note: 1 denotes the
170 // initial grid, so 2 would be subdividing the squares once)
171 // - iterationPixels: maximum number of pixels to subdivide in each integration iteration
172 // (max number of pixels that will be integrates is then 5*iterationPixels)
173 // - initialSubToFinal: once we decide to subdivide a square, do we automatically keep subdividing it
174 // until we reach maxSubdivision?
175 // - valdiag: the "value" and "distance" Diagnostic to use
176 SquareSubdivisionMesh(largecounter maxPixels, largecounter initialPixels, int maxSubdivide, largecounter iterationPixels, bool initialSubToFinal,
177 DiagBitflag valdiag)
178 : m_InitialPixels{static_cast<pixelcoord>(sqrt(initialPixels)) * static_cast<pixelcoord>(sqrt(initialPixels))},
179 m_MaxSubdivide{maxSubdivide},
180 m_RowColumnSize{static_cast<pixelcoord>((sqrt(initialPixels) - 1) * ExpInt(2, maxSubdivide - 1) + 1)},
181 m_PixelsLeft{maxPixels}, m_MaxPixels{maxPixels}, m_InfinitePixels{maxPixels == 0}, m_IterationPixels{iterationPixels},
182 m_InitialSubDividideToFinal{initialSubToFinal}, Mesh(valdiag)
